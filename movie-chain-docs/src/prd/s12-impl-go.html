12.1 Tenant context

An unexported key type makes it impossible for another package to write to this context slot by accident — a small detail that removes a whole class of confusion.

internal/tenancy/context.go
package tenancy

type TenantID string

type Context struct {
	TenantID    TenantID
	Tier        string
	UserID      string
	Role        string
	HoldSeconds int
	MaxSeats    int
}

// unexported key type: no other package can collide with or overwrite it
type ctxKey struct{}

func With(ctx context.Context, tc Context) context.Context {
	return context.WithValue(ctx, ctxKey{}, tc)
}

// FromContext returns an error rather than a zero value. A missing tenant
// is a bug; returning TenantID("") would silently query nothing — or,
// with a privileged role, everything.
func FromContext(ctx context.Context) (Context, error) {
	tc, ok := ctx.Value(ctxKey{}).(Context)
	if !ok || tc.TenantID == "" {
		return Context{}, ErrNoTenantContext
	}
	return tc, nil
}

// MustFromContext is for code paths that cannot proceed without a tenant.
func MustFromContext(ctx context.Context) Context {
	tc, err := FromContext(ctx)
	if err != nil { panic("tenancy: " + err.Error()) }
	return tc
}
internal/tenancy/middleware.go
func Middleware(reg *Registry, aud *audit.Logger) gin.HandlerFunc {
	return func(c *gin.Context) {
		claims, ok := auth.ClaimsFrom(c)      // set by AuthMiddleware
		if !ok || claims.TenantID == "" {
			c.AbortWithStatusJSON(401, gin.H{"error": "no_tenant_context"})
			return
		}

		// Guards STAFF endpoints only. A consumer token legitimately has no
		// tenant claim (§4.2) and goes through ConsumerOnly instead.
		// A tenant named in the path is a hint. The token is the authority.
		if hinted := c.Param("tenantSlug"); hinted != "" && hinted != claims.TenantSlug {
			aud.SecurityEvent(c, "tenant_mismatch", claims.TenantID, hinted)
			c.AbortWithStatus(404)            // 404, not 403 — see §8.4
			return
		}

		cfg, err := reg.Get(c.Request.Context(), TenantID(claims.TenantID))
		if err != nil || cfg.Status != "active" {
			c.AbortWithStatus(403)
			return
		}

		tc := Context{
			TenantID: cfg.ID, Tier: cfg.Tier, UserID: claims.Subject,
			Role: claims.Role, HoldSeconds: cfg.HoldSeconds, MaxSeats: cfg.MaxSeats,
		}
		// Replace the request context — everything downstream inherits it.
		c.Request = c.Request.WithContext(With(c.Request.Context(), tc))
		c.Next()
	}
}

12.2 The consumer path — no tenant claim at all

Consumer endpoints cannot use the middleware above: a consumer token deliberately carries no tenant, and a guest carries no token at all. The tenant is derived from the resource instead.

internal/tenancy/consumer.go
// ConsumerOnly guards the public booking surface. It accepts an
// anonymous request, accepts a consumer token, and REFUSES a staff
// token — that would let a Nyati clerk transact as a Prestige customer.
func ConsumerOnly(aud *audit.Logger) gin.HandlerFunc {
	return func(c *gin.Context) {
		claims, ok := auth.ClaimsFrom(c)
		if ok && claims.TenantID != "" {
			aud.SecurityEvent(c, "wrong_token_type", claims.TenantID, c.FullPath())
			c.AbortWithStatusJSON(403, gin.H{"error": "wrong_token_type"})
			return
		}
		// Guests are fine: consumerID stays empty. Ownership of a seat is
		// proved by the signed hold token, not by identity.
		var consumerID string
		if ok { consumerID = claims.Subject }
		c.Request = c.Request.WithContext(
			context.WithValue(c.Request.Context(), consumerKey{}, consumerID))
		c.Next()
	}
}
internal/inventory/handler.go — where the tenant actually comes from
func (h *Handler) CreateHold(c *gin.Context) {
	ctx := c.Request.Context()
	screeningID := c.Param("id")

	// The caller supplied an object id. WE supply the tenant, by reading
	// our own row. The caller's opinion of the tenant is never consulted —
	// note there is nowhere in this handler for such a value to enter.
	scr, err := h.catalogue.PublicScreening(ctx, screeningID)
	if err != nil { c.AbortWithStatus(404); return }

	cfg, err := h.reg.Get(ctx, scr.TenantID)
	if err != nil { c.AbortWithStatus(503); return }

	// From here down it is byte-identical to the staff path.
	ctx = tenancy.With(ctx, tenancy.Context{
		TenantID: cfg.ID, Tier: cfg.Tier, Role: "consumer",
		HoldSeconds: cfg.HoldSeconds, MaxSeats: cfg.MaxSeats,
	})

	var req HoldRequest
	if err := c.ShouldBindJSON(&req); err != nil { c.AbortWithStatus(400); return }

	hold, err := h.svc.Acquire(ctx, screeningID, req.SeatIDs)
	// ... 409 on conflict, 201 with the signed hold_token otherwise
}
Why this is not “trusting the URL”

Chapter 6 of the masterclass forbids taking the tenant from the request. This does not: the caller supplies screeningID, an object identifier, and the server derives the tenant from its own row. What would be forbidden is accepting a tenant_id supplied alongside it.

Go makes the safety of this unusually visible. Because tenancy.With returns a new ctx and every store method takes ctx, a reviewer can see in one screen exactly where the tenant entered and that nothing downstream can change it.

12.3 Data access and isolation

The critical subtlety in Go: SET LOCAL must execute on the same physical connection as the subsequent queries. With a pool, that means acquiring a connection explicitly and holding it for the transaction rather than letting the pool hand out a fresh one per statement.

internal/tenancy/db.go
type DB struct {
	pools map[string]*pgxpool.Pool   // "__pooled__" plus one per silo
	mu    sync.RWMutex
	reg   *Registry
}

// WithTenant runs fn inside a transaction scoped to the caller's tenant.
// It is the ONLY way the application reaches the database.
func (d *DB) WithTenant(ctx context.Context, fn func(pgx.Tx) error) error {
	tc, err := FromContext(ctx)
	if err != nil { return err }          // fail closed

	cfg, err := d.reg.Get(ctx, tc.TenantID)
	if err != nil { return err }

	// Pooled tenants share one pool; siloed tenants get their own.
	// Callers cannot tell which they received — and must not need to.
	pool := d.poolFor(cfg.DBRef)

	conn, err := pool.Acquire(ctx)             // hold ONE connection
	if err != nil { return err }
	defer conn.Release()

	tx, err := conn.Begin(ctx)
	if err != nil { return err }
	defer tx.Rollback(ctx)                    // no-op after Commit

	// SET LOCAL: scoped to this transaction. A plain SET would persist
	// on the pooled connection and leak into the next request.
	if _, err := tx.Exec(ctx, "SET LOCAL app.tenant_id = $1", string(tc.TenantID)); err != nil {
		return err
	}
	if err := fn(tx); err != nil { return err }
	return tx.Commit(ctx)
}
The Go-specific trap

pool.Exec(ctx, …) acquires an arbitrary connection from the pool, runs one statement, and releases it. If SET LOCAL is issued that way, it applies to a connection that is immediately returned to the pool — and the query that follows runs on a different connection with no tenant set. Under RLS that returns zero rows, which developers then “fix” by connecting as a privileged role, at which point isolation is gone entirely.

Always Acquire a connection and hold it. Forbid direct pool.Query / pool.Exec outside internal/tenancy with a lint rule, and write a test that proves a query outside WithTenant returns no rows.

12.4 The hold service

internal/inventory/hold.go
var acquireScript = redis.NewScript(`
  for i = 1, #KEYS do
    if redis.call('EXISTS', KEYS[i]) == 1 then return {0, i} end
  end
  for i = 1, #KEYS do
    redis.call('SET', KEYS[i], ARGV[1], 'NX', 'EX', ARGV[2])
  end
  return {1, 0}`)

func lockKey(t tenancy.TenantID, screening, seat string) string {
	return fmt.Sprintf("lock:t:%s:scr:%s:seat:%s", t, screening, seat)  // tenant first
}

func (s *Service) Acquire(ctx context.Context, screeningID string, seatIDs []string) (*Hold, error) {
	tc := tenancy.MustFromContext(ctx)
	if len(seatIDs) > tc.MaxSeats {
		return nil, ErrTooManySeats
	}

	holdID := uuid.New().String()
	keys := make([]string, len(seatIDs))
	for i, id := range seatIDs {
		keys[i] = lockKey(tc.TenantID, screeningID, id)
	}

	// 1 · Atomic across every seat — all or nothing.
	res, err := acquireScript.Run(ctx, s.rdb, keys, holdID, tc.HoldSeconds).Slice()
	if err != nil { return nil, err }
	if res[0].(int64) == 0 {
		return nil, &ConflictError{SeatID: seatIDs[res[1].(int64)-1]}
	}

	expiresAt := time.Now().Add(time.Duration(tc.HoldSeconds) * time.Second)

	// 2 · Durable record. The composite primary key is the real guarantee:
	//     if Redis ever admits two winners, Postgres rejects the second.
	err = s.db.WithTenant(ctx, func(tx pgx.Tx) error {
		return s.q.WithTx(tx).InsertSeatLocks(ctx, db.InsertSeatLocksParams{
			ScreeningID: screeningID, SeatIds: seatIDs,
			TenantID: string(tc.TenantID), HoldID: holdID, ExpiresAt: expiresAt,
		})
	})
	if err != nil {
		s.rdb.Del(ctx, keys...)      // never strand the seats
		return nil, err
	}
	return &Hold{ID: holdID, ExpiresAt: expiresAt}, nil
}

12.5 Auditable SQL with sqlc

The advantage of generating Go from SQL is that the tenant scoping sits in a reviewable file rather than behind a builder. Note that these queries do not carry a tenant_id predicate — RLS supplies it. That is deliberate, and the comment says so, because a reviewer's first instinct is to assume it was forgotten.

db/queries/booking.sql
-- name: GetBooking :one
-- No tenant_id predicate here. Row-level security adds it from
-- app.tenant_id, set by tenancy.DB.WithTenant. Adding it again would
-- duplicate the boundary in application code and hide RLS failures
-- from our tests. See PRD §6.4.
SELECT * FROM booking WHERE id = $1;

-- name: InsertSeatLocks :exec
INSERT INTO seat_lock (screening_id, seat_id, tenant_id, status, hold_id, expires_at)
SELECT @screening_id, unnest(@seat_ids::uuid[]), @tenant_id, 'held', @hold_id, @expires_at
ON CONFLICT (screening_id, seat_id) DO NOTHING;

-- name: ConfirmSeatLocks :many
UPDATE seat_lock
   SET status = 'sold', expires_at = NULL, booking_id = @booking_id
 WHERE hold_id = @hold_id AND status = 'held'
RETURNING seat_id;

12.6 Workers and the sweeper

Workers have no request, therefore no tenant in context

An asynq handler receives a bare context.Context. FromContext will return ErrNoTenantContext, which is correct fail-closed behaviour. Every job payload must carry its tenant_id, and every handler must call tenancy.With before touching data.

internal/worker/notify.go
func (h *NotifyHandler) ProcessTask(ctx context.Context, t *asynq.Task) error {
	var p NotifyPayload
	if err := json.Unmarshal(t.Payload(), &p); err != nil { return err }

	cfg, err := h.reg.Get(ctx, tenancy.TenantID(p.TenantID))
	if err != nil { return err }

	// Establish context from the payload — exactly as the middleware
	// does from the token. Everything below is now tenant-scoped.
	ctx = tenancy.With(ctx, tenancy.Context{
		TenantID: cfg.ID, Tier: cfg.Tier, Role: "system",
	})

	return h.db.WithTenant(ctx, func(tx pgx.Tx) error {
		b, err := h.q.WithTx(tx).GetBooking(ctx, p.BookingID)  // RLS scopes it
		if err != nil { return err }
		return h.sms.Send(ctx, b)
	})
}

The sweeper of §7.5 is the deliberate exception: it runs platform-wide with an elevated role and no tenant scope. There are exactly two such jobs in the system — the sweeper and the metering aggregator — and both live in a separate package, are reviewed as security-sensitive, and are covered by tests asserting they touch nothing beyond their remit.
